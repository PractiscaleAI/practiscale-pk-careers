export function generateStaticParams() {
  return routes
    .filter((x) => x !== "/")
    .map((x) => ({ slug: x.slice(1).split("/") }));
}
export async function generateMetadata({
  params,
}: {
  params: Promise<{ slug: string[] }>;
}): Promise<Metadata> {
  const { slug } = await params;
  const key = slug.join("/"),
    p = pages[key];
  return p
    ? {
        title:
          p.label === "Training & hiring path"
            ? "Appointment Setter Program"
            : key.includes("virtual-assistant")
              ? "Virtual Assistant Program for Women"
              : p.label,
        description: p.meta,
        alternates: { canonical: `/${key}` },
      }
    : { title: "Page not found" };
}
